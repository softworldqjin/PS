조건에 맞는 개발자 찾기
>
select  DISTINCT # Python과 C# 모두 할줄알아 중복되면 DISTINCT로 중복제거
        d.ID,
        d.EMAIL,
        d.FIRST_NAME,
        d.LAST_NAME
from DEVELOPERS d
inner join SKILLCODES s on d.SKILL_CODE & s.CODE ## 비트연산해서 0이 아닌(true)결과인 갖고있는 스킬들이 행으로 추가됨, 예를 들어 ID 1 C++추가, ID 1 Python 추가, ID 1 C# 추가
                                                 ## 여기서 같은 ID 여러개 중복됨
where s.NAME in ('Python', 'C#') ## 한 사람 당 추가된 여러 스킬들 중에 Python이나 C#만 남김
order by d.ID ASC;

JOIN 해서 특정 컬럼 세기 COUNT(컬럼명) 
>
select  e1.ID,
        count(e2.ID) as 'CHILD_COUNT' <<<< 밑에서 GROUP BY로 e1테이블의 ID컬럼 데이터값 같은거끼리 묶었는데, JOIN해온 e2테이블의 ID컬럼의 존재개수를 센다! NULL일 경우 카운트 안함!
from ECOLI_DATA e1 
left join ECOLI_DATA e2 on e1.ID = e2.PARENT_ID
group by e1.ID
order by e1.ID ASC

CASE문
>
CASE
    WHEN 조건1 THEN 결과1
    WHEN 조건2 THEN 결과2
    ELSE 결과3
END as 별칭

IFNULL(값, 대체값) vs NULLIF(값1, 값2)
>
IFNULL(값, 대체값) -> 값이 NULL이면 대체값 반환
NULLIF(값1, 값2)   -> 값1 과 값2 같으면 NULL 반환

CASE NTILE(나눌개수) OVER (order by 컬럼명 DESC 또는 ASC))
4개 나누면 0~25퍼 26~50퍼, 51~75퍼, 76~100퍼 로 나뉘어서 정해준 값 들어감
>
select  ID,
        CASE NTILE(4) OVER (order by SIZE_OF_COLONY DESC)
            when 1 then 'CRITICAL'
            when 2 then 'HIGH'
            when 3 then 'MEDIUM'
            when 4 then 'LOW'
        end as 'COLONY_NAME'
from ECOLI_DATA
order by ID ASC;

SELECT * FROM TABLE1
WHERE ~~
UNION

서브 쿼리 JOIN
>
서브쿼리로 만든 가상 테이블의 컬럼 = SELECT 절에 명시한 컬럼만 존재한다.

select *
from FOOD_PRODUCT f
join (
select CATEGORY, MAX(PRICE) as MAXPRICE
from FOOD_PRODUCT
group by CATEGORY
) m
on f.CATEGORY = m.CATEGORY

JOIN → 컬럼은 전부 합쳐짐
ON → 조건 맞는 행만 결합
SELECT → 합쳐진 테이블에서 최종적으로 보여줄 컬럼만 선택
